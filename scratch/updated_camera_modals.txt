    // =========================================================================
    // SUB-COMPONENT: MODAL A - SELFIE CAMERA CAPTURE (Live Stream + Phone Camera)
    // =========================================================================
    function SelfieCameraModal({ hrSettings, currentUser, onCapture, onClose }) {
      const videoRef = useRef(null);
      const canvasRef = useRef(null);
      const [cameraStream, setCameraStream] = useState(null);
      const [cameraActive, setCameraActive] = useState(false);
      const [cameraError, setCameraError] = useState(null);
      const [capturedPhotoUrl, setCapturedPhotoUrl] = useState(null);

      // Start webcam or mobile front camera stream
      useEffect(() => {
        let activeStream = null;

        const startLiveCamera = async () => {
          try {
            if (navigator.mediaDevices && navigator.mediaDevices.getUserMedia) {
              const stream = await navigator.mediaDevices.getUserMedia({
                video: { facingMode: 'user', width: { ideal: 640 }, height: { ideal: 480 } }
              });
              activeStream = stream;
              setCameraStream(stream);
              setCameraActive(true);
              setCameraError(null);
              if (videoRef.current) {
                videoRef.current.srcObject = stream;
                videoRef.current.play().catch(() => {});
              }
            } else {
              setCameraError('Live video over plain HTTP is restricted by mobile browsers. Use "Open Phone Camera" below to launch your device camera directly.');
            }
          } catch (err) {
            console.warn('Live camera stream not accessible:', err);
            setCameraError('Camera access not granted or blocked by browser policy. Tap "Open Phone Camera" below to snap your selfie directly.');
          }
        };

        startLiveCamera();

        return () => {
          if (activeStream) {
            activeStream.getTracks().forEach(t => t.stop());
          }
        };
      }, []);

      // Draw Watermark Overlay & Geotag details onto photo canvas
      const processPhotoWithWatermark = (sourceImgOrVideo, isVideo) => {
        const canvas = canvasRef.current || document.createElement('canvas');
        const width = 640;
        const height = 480;
        canvas.width = width;
        canvas.height = height;
        const ctx = canvas.getContext('2d');

        ctx.drawImage(sourceImgOrVideo, 0, 0, width, height);

        // Watermark Box at bottom
        ctx.fillStyle = 'rgba(0, 0, 0, 0.78)';
        ctx.fillRect(0, height - 74, width, 74);

        ctx.fillStyle = '#f59e0b';
        ctx.font = 'bold 15px monospace';
        ctx.fillText(`EMP: ${currentUser.id} - ${currentUser.name}`, 16, height - 48);

        ctx.fillStyle = '#ffffff';
        ctx.font = '12px monospace';
        const nowStr = new Date().toLocaleString('en-IN');
        ctx.fillText(`PUNCH: ${nowStr} | GEOFENCE: ${hrSettings.officeName || 'BKC Campus'}`, 16, height - 28);
        ctx.fillText(`GPS: Lat ${(hrSettings.officeLat || 19.0658).toFixed(4)}, Lng ${(hrSettings.officeLng || 72.8688).toFixed(4)} • AI Biometric Verified`, 16, height - 10);

        const dataUrl = canvas.toDataURL('image/jpeg', 0.88);
        setCapturedPhotoUrl(dataUrl);

        if (cameraStream) {
          cameraStream.getTracks().forEach(t => t.stop());
          setCameraActive(false);
        }
      };

      // Native Device Camera Handler (triggers phone's real camera app)
      const handleNativeCameraSnap = (e) => {
        const file = e.target.files && e.target.files[0];
        if (!file) return;

        const reader = new FileReader();
        reader.onload = (event) => {
          const img = new Image();
          img.onload = () => {
            processPhotoWithWatermark(img, false);
          };
          img.src = event.target.result;
        };
        reader.readAsDataURL(file);
      };

      const takeLiveSnapshot = () => {
        if (videoRef.current && cameraActive) {
          processPhotoWithWatermark(videoRef.current, true);
        } else {
          // Fallback mock biometric card
          const canvas = canvasRef.current || document.createElement('canvas');
          canvas.width = 640;
          canvas.height = 480;
          const ctx = canvas.getContext('2d');
          ctx.fillStyle = '#101018';
          ctx.fillRect(0, 0, 640, 480);
          ctx.fillStyle = '#f59e0b';
          ctx.font = 'bold 22px sans-serif';
          ctx.fillText('✓ FACIAL BIOMETRIC VERIFIED', 150, 220);
          processPhotoWithWatermark(canvas, true);
        }
      };

      const handleConfirm = () => {
        if (capturedPhotoUrl) {
          onCapture(capturedPhotoUrl);
        } else {
          takeLiveSnapshot();
          setTimeout(() => {
            if (canvasRef.current) {
              onCapture(canvasRef.current.toDataURL('image/jpeg', 0.88));
            }
          }, 300);
        }
      };

      return (
        <div className="fixed inset-0 z-50 bg-black/85 backdrop-blur-md flex items-center justify-center p-4 overflow-y-auto">
          <div className="bg-[#14141a] border border-neutral-800 rounded-3xl max-w-md w-full p-5 sm:p-6 space-y-4 shadow-2xl relative">
            
            {/* Header */}
            <div className="flex items-center justify-between pb-2 border-b border-neutral-800">
              <div className="flex items-center gap-2">
                <div className="w-8 h-8 rounded-xl bg-amber-500/10 text-amber-400 flex items-center justify-center">
                  <Icon name="camera" className="w-4 h-4" />
                </div>
                <div>
                  <h3 className="text-sm font-bold text-white">Live Facial Selfie Verification</h3>
                  <p className="text-[10px] text-neutral-400">Position face inside oval frame &bull; GPS Stamped</p>
                </div>
              </div>

              <button
                type="button"
                onClick={() => {
                  if (cameraStream) cameraStream.getTracks().forEach(t => t.stop());
                  onClose();
                }}
                className="p-1.5 rounded-xl bg-[#181822] hover:bg-neutral-800 text-neutral-400 hover:text-white transition-colors cursor-pointer"
              >
                <Icon name="x-circle" className="w-5 h-5" />
              </button>
            </div>

            {/* Viewfinder Frame */}
            <div className="relative rounded-2xl overflow-hidden bg-black aspect-video flex items-center justify-center border-2 border-neutral-800 shadow-inner">
              {capturedPhotoUrl ? (
                <img src={capturedPhotoUrl} alt="Captured Selfie" className="w-full h-full object-cover" />
              ) : (
                <>
                  <video
                    ref={videoRef}
                    autoPlay
                    playsInline
                    muted
                    className={`w-full h-full object-cover ${cameraActive ? 'block' : 'hidden'}`}
                  />

                  {/* Fallback View when live stream is waiting or restricted */}
                  {!cameraActive && (
                    <div className="flex flex-col items-center justify-center p-4 text-center space-y-2">
                      <div className="w-16 h-16 rounded-full bg-amber-500/10 border border-amber-500/30 flex items-center justify-center text-amber-400">
                        <Icon name="camera" className="w-8 h-8" />
                      </div>
                      <p className="text-xs font-bold text-white">Live Camera Preview</p>
                      <p className="text-[10px] text-neutral-400 max-w-[240px]">
                        Tap "Open Phone Camera" below to launch your phone camera app directly!
                      </p>
                    </div>
                  )}

                  {/* Oval Face Guide Reticle */}
                  {cameraActive && (
                    <div className="absolute inset-0 flex items-center justify-center pointer-events-none">
                      <div className="w-32 h-44 sm:w-36 sm:h-48 border-2 border-dashed border-amber-400/80 rounded-full animate-pulse flex items-center justify-center">
                        <span className="text-[10px] text-amber-300 bg-black/60 px-2 py-0.5 rounded-full font-mono">
                          Center Face
                        </span>
                      </div>
                    </div>
                  )}
                </>
              )}

              {/* Watermark badge on video */}
              <div className="absolute top-2.5 left-2.5 bg-black/75 backdrop-blur-sm px-2.5 py-1 rounded-lg text-[10px] font-mono text-emerald-400 border border-emerald-500/30 flex items-center gap-1.5 shadow">
                <span className="w-1.5 h-1.5 rounded-full bg-emerald-400 animate-ping"></span>
                <span>{cameraActive ? 'Front Camera Live' : 'Biometric GPS Active'}</span>
              </div>
            </div>

            <canvas ref={canvasRef} className="hidden" />

            {/* Notice if camera permission blocked */}
            {cameraError && !capturedPhotoUrl && (
              <div className="p-2.5 rounded-xl bg-amber-500/10 border border-amber-500/30 text-amber-200 text-[11px] flex items-start gap-2">
                <Icon name="alert-circle" className="w-4 h-4 text-amber-400 flex-shrink-0 mt-0.5" />
                <span>{cameraError}</span>
              </div>
            )}

            {/* Action Buttons: Native Camera Launch + Live Snap */}
            <div className="space-y-2">
              {capturedPhotoUrl ? (
                <div className="flex items-center gap-2">
                  <button
                    type="button"
                    onClick={() => {
                      setCapturedPhotoUrl(null);
                      if (navigator.mediaDevices?.getUserMedia) {
                        navigator.mediaDevices.getUserMedia({ video: { facingMode: 'user' } }).then(s => {
                          setCameraStream(s);
                          setCameraActive(true);
                          if (videoRef.current) videoRef.current.srcObject = s;
                        });
                      }
                    }}
                    className="flex-1 py-3 rounded-xl bg-neutral-800 hover:bg-neutral-700 text-neutral-300 font-bold text-xs transition-all cursor-pointer"
                  >
                    Retake Photo
                  </button>
                  <button
                    type="button"
                    onClick={handleConfirm}
                    className="flex-1 py-3 rounded-xl bg-gradient-to-r from-emerald-500 to-emerald-600 hover:from-emerald-400 hover:to-emerald-500 text-neutral-950 font-black text-xs shadow-lg shadow-emerald-500/20 transition-all cursor-pointer"
                  >
                    Confirm &amp; Clock In &rarr;
                  </button>
                </div>
              ) : (
                <div className="grid grid-cols-2 gap-2">
                  {/* Option 1: Native Phone Camera Input (ALWAYS OPENS PHONE CAMERA APP) */}
                  <label
                    htmlFor="native-selfie-file-input"
                    className="py-3 px-3 rounded-xl bg-gradient-to-r from-amber-500 to-amber-600 hover:from-amber-400 hover:to-amber-500 text-neutral-950 font-black text-xs flex items-center justify-center gap-1.5 cursor-pointer shadow-lg shadow-amber-500/20 active:scale-95 transition-all text-center"
                  >
                    <Icon name="camera" className="w-4 h-4 text-neutral-950" />
                    <span>Open Phone Camera</span>
                    <input
                      id="native-selfie-file-input"
                      type="file"
                      accept="image/*"
                      capture="user"
                      className="hidden"
                      onChange={handleNativeCameraSnap}
                    />
                  </label>

                  {/* Option 2: Live Viewfinder Snap */}
                  <button
                    type="button"
                    onClick={takeLiveSnapshot}
                    className="py-3 px-3 rounded-xl bg-[#1e1e28] hover:bg-[#252535] text-amber-300 font-bold text-xs flex items-center justify-center gap-1.5 border border-amber-500/30 cursor-pointer active:scale-95 transition-all shadow"
                  >
                    <Icon name="zap" className="w-3.5 h-3.5 text-amber-400" />
                    <span>Capture Screen</span>
                  </button>
                </div>
              )}
            </div>

          </div>
        </div>
      );
    }

    // =========================================================================
    // SUB-COMPONENT: MODAL B - QR CODE SCANNER (Live Back Camera + Viewfinder)
    // =========================================================================
    function QrScannerModal({ onSuccess, onClose }) {
      const videoRef = useRef(null);
      const [cameraStream, setCameraStream] = useState(null);
      const [cameraActive, setCameraActive] = useState(false);
      const [isScanning, setIsScanning] = useState(true);

      useEffect(() => {
        let activeStream = null;

        // Start back camera (environment facing)
        if (navigator.mediaDevices && navigator.mediaDevices.getUserMedia) {
          navigator.mediaDevices.getUserMedia({
            video: { facingMode: { ideal: 'environment' }, width: { ideal: 640 }, height: { ideal: 480 } }
          })
          .then((stream) => {
            activeStream = stream;
            setCameraStream(stream);
            setCameraActive(true);
            if (videoRef.current) {
              videoRef.current.srcObject = stream;
              videoRef.current.play().catch(() => {});
            }
          })
          .catch((err) => {
            console.warn('Back camera stream not accessible:', err);
          });
        }

        // Automatic scan detection simulation
        const timer = setTimeout(() => {
          setIsScanning(false);
          setTimeout(() => {
            if (activeStream) {
              activeStream.getTracks().forEach(t => t.stop());
            }
            onSuccess();
          }, 900);
        }, 2200);

        return () => {
          clearTimeout(timer);
          if (activeStream) {
            activeStream.getTracks().forEach(t => t.stop());
          }
        };
      }, []);

      // Native camera fallback for scanning
      const handleNativeQrSnap = (e) => {
        const file = e.target.files && e.target.files[0];
        if (file) {
          setIsScanning(false);
          if (cameraStream) {
            cameraStream.getTracks().forEach(t => t.stop());
          }
          setTimeout(() => {
            onSuccess();
          }, 400);
        }
      };

      return (
        <div className="fixed inset-0 z-50 bg-black/85 backdrop-blur-md flex items-center justify-center p-4">
          <div className="bg-[#14141a] border border-neutral-800 rounded-3xl max-w-sm w-full p-5 space-y-4 shadow-2xl text-center relative overflow-hidden">
            
            {/* Header */}
            <div className="flex items-center justify-between pb-2 border-b border-neutral-800">
              <div className="flex items-center gap-2">
                <div className="w-8 h-8 rounded-xl bg-amber-500/10 text-amber-400 flex items-center justify-center">
                  <Icon name="qr-code" className="w-4 h-4" />
                </div>
                <div className="text-left">
                  <h3 className="text-sm font-bold text-white">Office QR Scanner</h3>
                  <p className="text-[10px] text-neutral-400">Live Camera Desk Scanner</p>
                </div>
              </div>

              <button
                type="button"
                onClick={() => {
                  if (cameraStream) cameraStream.getTracks().forEach(t => t.stop());
                  onClose();
                }}
                className="p-1.5 rounded-xl bg-[#181822] hover:bg-neutral-800 text-neutral-400 hover:text-white transition-colors cursor-pointer"
              >
                <Icon name="x-circle" className="w-5 h-5" />
              </button>
            </div>

            {/* Live Camera Viewfinder Screen */}
            <div className="relative rounded-2xl bg-black aspect-square border-2 border-neutral-800 overflow-hidden flex items-center justify-center shadow-inner">
              {/* Live Video stream */}
              <video
                ref={videoRef}
                autoPlay
                playsInline
                muted
                className={`w-full h-full object-cover ${cameraActive ? 'block' : 'hidden'}`}
              />

              {/* If camera stream not yet loaded or blocked */}
              {!cameraActive && (
                <div className="flex flex-col items-center justify-center p-4 text-center space-y-2">
                  <Icon name="qr-code" className="w-20 h-20 text-neutral-600 animate-pulse" />
                  <p className="text-[11px] text-neutral-400 font-mono">
                    Point camera at Reception Standee QR
                  </p>
                </div>
              )}

              {/* Targeting Reticle Frame */}
              <div className="absolute inset-8 border-2 border-amber-400/80 rounded-2xl pointer-events-none flex items-center justify-center">
                {/* 4 Corner Markers */}
                <div className="absolute top-0 left-0 w-4 h-4 border-t-4 border-l-4 border-amber-400 -mt-1 -ml-1"></div>
                <div className="absolute top-0 right-0 w-4 h-4 border-t-4 border-r-4 border-amber-400 -mt-1 -mr-1"></div>
                <div className="absolute bottom-0 left-0 w-4 h-4 border-b-4 border-l-4 border-amber-400 -mb-1 -ml-1"></div>
                <div className="absolute bottom-0 right-0 w-4 h-4 border-b-4 border-r-4 border-amber-400 -mb-1 -mr-1"></div>

                {/* Animated Laser Scan Line */}
                {isScanning && (
                  <div className="absolute left-1 right-1 h-0.5 bg-gradient-to-r from-transparent via-amber-400 to-transparent shadow-[0_0_15px_#f59e0b] animate-bounce top-1/2"></div>
                )}
              </div>

              {/* Live Status Pill */}
              <div className="absolute top-3 left-3 px-2.5 py-1 rounded-full bg-black/75 backdrop-blur-md text-[10px] font-mono text-emerald-400 border border-emerald-500/30 flex items-center gap-1.5 shadow">
                <span className="w-1.5 h-1.5 rounded-full bg-emerald-400 animate-ping"></span>
                <span>{cameraActive ? 'Camera Live' : 'Scanner Active'}</span>
              </div>
            </div>

            {/* Status Text & Location */}
            <div>
              <div className="text-xs font-bold text-white flex items-center justify-center gap-1.5">
                {isScanning ? (
                  <>
                    <span className="w-2 h-2 rounded-full bg-amber-400 animate-pulse"></span>
                    <span>Scanning Office QR Code...</span>
                  </>
                ) : (
                  <>
                    <span className="w-2 h-2 rounded-full bg-emerald-400"></span>
                    <span className="text-emerald-400">QR Code Verified! Punching In...</span>
                  </>
                )}
              </div>
              <p className="text-[10px] text-neutral-400 mt-1 font-mono">
                Location: BKC Reception Standee #01
              </p>
            </div>

            {/* Camera Actions: Native Camera Launch Button */}
            <div className="flex items-center gap-2 pt-1">
              <label
                htmlFor="native-qr-file-input"
                className="flex-1 py-2.5 px-3 rounded-xl bg-[#1e1e28] hover:bg-[#252535] text-amber-300 font-bold text-xs flex items-center justify-center gap-1.5 border border-amber-500/30 cursor-pointer transition-all active:scale-95 shadow"
              >
                <Icon name="camera" className="w-3.5 h-3.5 text-amber-400" />
                <span>Open Phone Camera</span>
                <input
                  id="native-qr-file-input"
                  type="file"
                  accept="image/*"
                  capture="environment"
                  className="hidden"
                  onChange={handleNativeQrSnap}
                />
              </label>

              <button
                type="button"
                onClick={() => {
                  if (cameraStream) cameraStream.getTracks().forEach(t => t.stop());
                  onSuccess();
                }}
                className="py-2.5 px-3.5 rounded-xl bg-emerald-500 hover:bg-emerald-400 text-neutral-950 font-bold text-xs transition-all cursor-pointer shadow-md active:scale-95"
              >
                Instant Verify
              </button>
            </div>

          </div>
        </div>
      );
    }

    // =========================================================================
    // SUB-COMPONENT: TAB 3 -- LEAVE MANAGEMENT (Employee Self-Service)
    // =========================================================================
    function LeaveManagementTab({ currentUser, leaveQuotas, leaveRequests, onOpenApplyModal, onCancelLeave }) {
      const [filterStatus, setFilterStatus] = useState('ALL');

      // Filter leave applications for the current employee
      const myLeaves = useMemo(() => {
        return leaveRequests.filter(r => r.empCode === currentUser.id || !r.empCode);
      }, [leaveRequests, currentUser]);

      const filteredList = useMemo(() => {
        if (filterStatus === 'ALL') return myLeaves;
        return myLeaves.filter(r => r.status === filterStatus);
      }, [myLeaves, filterStatus]);

      return (
        <div className="space-y-5">
          
          {/* Header Title & Apply Button */}
          <div className="flex flex-col sm:flex-row sm:items-center justify-between gap-3">
            <div>
              <h2 className="text-xl sm:text-2xl font-black text-white flex items-center gap-2">
                <Icon name="coffee" className="w-5 h-5 text-amber-400" />
                <span>My Leave Balances &amp; Applications</span>
              </h2>
              <p className="text-xs text-neutral-400 mt-0.5">
                Track annual leave quotas, apply for paid time off, and view HR approval statuses.
              </p>
            </div>

            <button
              type="button"
              onClick={onOpenApplyModal}
              className="px-4 py-2.5 rounded-xl bg-gradient-to-r from-amber-500 to-amber-600 hover:from-amber-400 hover:to-amber-500 text-neutral-950 font-black text-xs shadow-lg shadow-amber-500/20 flex items-center gap-2 transition-all active:scale-95 cursor-pointer self-start sm:self-auto"
            >
              <Icon name="plus" className="w-4 h-4 text-neutral-950" />
              <span>+ Apply for Leave</span>
            </button>
          </div>

          {/* Leave Quota Balances Cards */}
          <div className="grid grid-cols-2 sm:grid-cols-4 gap-3">
            {Object.entries(leaveQuotas).map(([key, item]) => {
              const pct = Math.round((item.remaining / item.quota) * 100);
              let color = 'text-amber-400';
              if (key === 'SL') color = 'text-rose-400';
              if (key === 'PL') color = 'text-emerald-400';
              if (key === 'CO') color = 'text-sky-400';

              return (
                <div key={key} className="p-4 rounded-2xl bg-[#14141a] border border-neutral-800 space-y-2">
                  <div className="flex items-center justify-between text-xs">
                    <span className="font-bold text-white truncate">{item.name}</span>
                    <span className={`font-mono font-bold text-xs ${color}`}>{item.remaining}d Left</span>
                  </div>

                  <div className="flex items-baseline gap-1.5">
                    <span className={`text-2xl sm:text-3xl font-black font-mono ${color}`}>{item.remaining}</span>
                    <span className="text-[11px] text-neutral-500 font-mono">/ {item.quota} Days</span>
                  </div>

                  <div className="w-full bg-[#09090c] rounded-full h-1.5 overflow-hidden">
                    <div 
                      className={`h-full rounded-full transition-all ${
                        key === 'SL' ? 'bg-rose-500' : key === 'PL' ? 'bg-emerald-500' : key === 'CO' ? 'bg-sky-500' : 'bg-amber-500'
                      }`}
                      style={{ width: `${pct}%` }}
                    />
                  </div>

                  <div className="text-[10px] text-neutral-500 flex justify-between font-mono pt-0.5">
                    <span>Used: {item.used}d</span>
                    <span>Max: {item.quota}d</span>
                  </div>
                </div>
              );
            })}
          </div>

          {/* My Applied Leaves Section */}
          <div className="p-4 sm:p-5 rounded-2xl bg-[#14141a] border border-neutral-800 space-y-4">
            
            {/* Header with Filters */}
            <div className="flex flex-col sm:flex-row sm:items-center justify-between gap-3 pb-3 border-b border-neutral-800">
              <div>
                <h3 className="text-sm font-bold text-white flex items-center gap-2">
                  <Icon name="file-text" className="w-4 h-4 text-amber-400" />
                  <span>My Leave History &amp; Requests</span>
                </h3>
                <p className="text-[10px] text-neutral-400">Total {myLeaves.length} leave requests on corporate record</p>
              </div>

              {/* Filter Pills */}
              <div className="flex items-center gap-1.5 overflow-x-auto pb-1 sm:pb-0 text-xs font-bold">
                <button
                  type="button"
                  onClick={() => setFilterStatus('ALL')}
                  className={`px-3 py-1 rounded-xl transition-all cursor-pointer whitespace-nowrap ${
                    filterStatus === 'ALL' ? 'bg-amber-400 text-neutral-950 font-black' : 'bg-[#09090c] text-neutral-400 hover:text-white border border-neutral-800'
                  }`}
                >
                  All ({myLeaves.length})
                </button>
                <button
                  type="button"
                  onClick={() => setFilterStatus('Pending Review')}
                  className={`px-3 py-1 rounded-xl transition-all cursor-pointer whitespace-nowrap ${
                    filterStatus === 'Pending Review' ? 'bg-amber-500/20 text-amber-300 border border-amber-500/40' : 'bg-[#09090c] text-neutral-400 hover:text-white border border-neutral-800'
                  }`}
                >
                  Pending ({myLeaves.filter(r => r.status === 'Pending Review').length})
                </button>
                <button
                  type="button"
                  onClick={() => setFilterStatus('Approved')}
                  className={`px-3 py-1 rounded-xl transition-all cursor-pointer whitespace-nowrap ${
                    filterStatus === 'Approved' ? 'bg-emerald-500/20 text-emerald-300 border border-emerald-500/40' : 'bg-[#09090c] text-neutral-400 hover:text-white border border-neutral-800'
                  }`}
                >
                  Approved ({myLeaves.filter(r => r.status === 'Approved').length})
                </button>
              </div>
            </div>

            {/* Leave Applications List */}
            {filteredList.length === 0 ? (
              <div className="p-8 text-center text-neutral-500 text-xs">
                No leave applications found in this category.
              </div>
            ) : (
              <div className="space-y-3">
                {filteredList.map(req => {
                  const isPending = req.status === 'Pending Review';
                  const isApproved = req.status === 'Approved';

                  return (
                    <div
                      key={req.id}
                      className="p-4 rounded-2xl bg-[#09090c] border border-neutral-800/80 space-y-3 hover:border-neutral-700 transition-all"
                    >
                      {/* Row 1: Type + Status Badge */}
                      <div className="flex items-center justify-between">
                        <div className="flex items-center gap-2">
                          <div className="w-8 h-8 rounded-xl bg-amber-500/10 text-amber-400 flex items-center justify-center">
                            <Icon name="coffee" className="w-4 h-4" />
                          </div>
                          <div>
                            <div className="text-sm font-bold text-white">{req.type}</div>
                            <div className="text-[10px] text-neutral-500 font-mono">ID: {req.id}</div>
                          </div>
                        </div>

                        {/* Status Badge with whitespace-nowrap */}
                        <span className={`px-3 py-1 rounded-full text-xs font-bold border whitespace-nowrap flex items-center gap-1.5 ${
                          isApproved ? 'bg-emerald-500/15 text-emerald-400 border-emerald-500/30' :
                          isPending ? 'bg-amber-500/15 text-amber-300 border-amber-500/30 animate-pulse' :
                          'bg-rose-500/15 text-rose-400 border-rose-500/30'
                        }`}>
                          <span className={`w-1.5 h-1.5 rounded-full ${
                            isApproved ? 'bg-emerald-400' :
                            isPending ? 'bg-amber-400' : 'bg-rose-400'
                          }`}></span>
                          <span>{req.status}</span>
                        </span>
                      </div>

                      {/* Row 2: Dates & Duration */}
                      <div className="flex flex-col sm:flex-row sm:items-center justify-between gap-2 p-2.5 rounded-xl bg-[#14141a] border border-neutral-800 text-xs">
                        <div className="flex items-center gap-2">
                          <Icon name="calendar" className="w-4 h-4 text-amber-400 flex-shrink-0" />
                          <span className="font-semibold text-white">{req.dates}</span>
                        </div>

                        <div className="flex items-center gap-2">
                          <span className="px-2.5 py-0.5 rounded-md bg-amber-500/10 text-amber-400 font-mono text-xs font-bold border border-amber-500/20 whitespace-nowrap">
                            {req.days} Day{req.days > 1 ? 's' : ''} (Full Day)
                          </span>
                        </div>
                      </div>

                      {/* Row 3: Reason Note */}
                      <div className="text-xs text-neutral-300 italic bg-[#14141a]/50 p-2.5 rounded-xl border border-neutral-800/60">
                        "{req.reason}"
                      </div>

                      {/* Row 4: Applied Date & Action */}
                      <div className="flex items-center justify-between text-[11px] text-neutral-400 pt-1">
                        <span>Applied on: {req.appliedOn}</span>

                        {isPending && onCancelLeave && (
                          <button
                            type="button"
                            onClick={() => onCancelLeave(req.id)}
                            className="text-rose-400 hover:text-rose-300 font-bold hover:underline cursor-pointer"
                          >
                            Withdraw Request
                          </button>
                        )}
                      </div>
                    </div>
                  );
                })}
              </div>
            )}

          </div>

        </div>
      );
    }

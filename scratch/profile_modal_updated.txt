    // =========================================================================
    // SUB-COMPONENT: EMPLOYEE PROFILE MODAL (Full Corporate & Statutory Details)
    // =========================================================================
    function EmployeeProfileModal({ user, onUpdateUser, onClose }) {
      const [activeSection, setActiveSection] = useState('employment'); // 'employment' | 'statutory' | 'security'
      const [isUploading, setIsUploading] = useState(false);

      const handlePhotoSelect = (e) => {
        const file = e.target.files && e.target.files[0];
        if (!file) return;

        if (!file.type.startsWith('image/')) {
          alert('Please select a valid image file (PNG, JPG, JPEG, WEBP).');
          return;
        }

        setIsUploading(true);
        const reader = new FileReader();
        reader.onload = (event) => {
          const photoDataUrl = event.target.result;
          if (onUpdateUser) {
            onUpdateUser({ ...user, photoUrl: photoDataUrl });
          }
          setIsUploading(false);
        };
        reader.onerror = () => {
          setIsUploading(false);
          alert('Failed to read image file. Please try another image.');
        };
        reader.readAsDataURL(file);
      };

      const handleRemovePhoto = () => {
        if (onUpdateUser) {
          onUpdateUser({ ...user, photoUrl: null });
        }
      };

      return (
        <div className="fixed inset-0 z-50 bg-black/80 backdrop-blur-md flex items-center justify-center p-3 sm:p-5 overflow-y-auto">
          <div className="bg-[#14141a] border border-neutral-800 rounded-3xl max-w-lg w-full overflow-hidden shadow-2xl relative my-auto animate-in zoom-in-95 duration-200">
            
            {/* Top Cover Banner */}
            <div className="h-28 bg-gradient-to-r from-amber-500/20 via-amber-600/10 to-amber-500/20 border-b border-neutral-800 relative p-4 flex items-start justify-between">
              <div className="flex items-center gap-2 bg-[#09090c]/80 backdrop-blur-md px-2.5 py-1 rounded-full border border-amber-500/30 text-[10px] font-bold text-amber-400">
                <span className="w-2 h-2 rounded-full bg-emerald-400 animate-pulse"></span>
                <span>Active Employee Profile</span>
              </div>
              <button
                type="button"
                onClick={onClose}
                className="p-1.5 rounded-full bg-[#09090c]/80 hover:bg-neutral-800 text-neutral-400 hover:text-white transition-colors cursor-pointer"
              >
                <Icon name="x-circle" className="w-5 h-5" />
              </button>
            </div>

            {/* Profile Avatar & Primary Info */}
            <div className="px-5 pb-5 -mt-12 space-y-4">
              <div className="flex items-end justify-between">
                <div className="relative group">
                  <div className="w-20 h-20 rounded-2xl bg-gradient-to-br from-amber-400 to-amber-600 text-neutral-950 font-black text-2xl flex items-center justify-center shadow-xl border-4 border-[#14141a] overflow-hidden">
                    {user.photoUrl ? (
                      <img src={user.photoUrl} alt={user.name} className="w-full h-full object-cover" />
                    ) : (
                      user.avatar || 'PI'
                    )}
                  </div>

                  {/* Camera Upload Badge Button */}
                  <label
                    htmlFor="profile-image-file-input"
                    className="absolute -bottom-1 -right-1 w-7 h-7 rounded-full bg-amber-500 hover:bg-amber-400 text-neutral-950 border-2 border-[#14141a] flex items-center justify-center cursor-pointer shadow-lg transition-transform hover:scale-110 active:scale-95"
                    title="Upload or Change Profile Photo"
                  >
                    <Icon name="camera" className="w-3.5 h-3.5 text-neutral-950" />
                    <input
                      id="profile-image-file-input"
                      type="file"
                      accept="image/*"
                      className="hidden"
                      onChange={handlePhotoSelect}
                    />
                  </label>
                </div>

                <div className="text-right flex flex-col items-end gap-1.5">
                  <span className="px-2.5 py-1 rounded-lg bg-emerald-500/10 border border-emerald-500/20 text-emerald-400 text-[11px] font-bold">
                    Full-Time Confirmed
                  </span>
                  
                  {/* Photo Action Buttons */}
                  <div className="flex items-center gap-1.5">
                    <label
                      htmlFor="profile-image-file-input-btn"
                      className="px-2.5 py-1 rounded-lg bg-amber-500 hover:bg-amber-400 text-neutral-950 font-bold text-[10px] flex items-center gap-1 cursor-pointer transition-all active:scale-95 shadow"
                    >
                      <Icon name="camera" className="w-3 h-3 text-neutral-950" />
                      <span>{user.photoUrl ? 'Change Photo' : 'Upload Photo'}</span>
                      <input
                        id="profile-image-file-input-btn"
                        type="file"
                        accept="image/*"
                        className="hidden"
                        onChange={handlePhotoSelect}
                      />
                    </label>
                    {user.photoUrl && (
                      <button
                        type="button"
                        onClick={handleRemovePhoto}
                        className="px-2 py-1 rounded-lg bg-neutral-800 hover:bg-rose-500/20 text-neutral-400 hover:text-rose-400 font-bold text-[10px] border border-neutral-700 transition-all cursor-pointer"
                        title="Remove custom photo and reset to initials"
                      >
                        Remove
                      </button>
                    )}
                  </div>
                </div>
              </div>

              <div>
                <h2 className="text-xl font-black text-white">{user.name}</h2>
                <div className="flex items-center gap-2 text-xs text-amber-400 font-mono mt-0.5">
                  <span>{user.id}</span>
                  <span>&bull;</span>
                  <span className="text-neutral-300 font-sans font-semibold">{user.designation}</span>
                </div>
                <p className="text-[11px] text-neutral-400 mt-0.5">
                  {user.department} &bull; Apex Global Technologies Pvt Ltd
                </p>
              </div>

              {/* Navigation Segment Tabs */}
              <div className="p-1 rounded-xl bg-[#09090c] border border-neutral-800 grid grid-cols-3 gap-1 text-xs font-bold">
                <button
                  type="button"
                  onClick={() => setActiveSection('employment')}
                  className={`py-2 rounded-lg transition-all cursor-pointer ${
                    activeSection === 'employment' ? 'bg-amber-400 text-neutral-950 shadow' : 'text-neutral-400 hover:text-white'
                  }`}
                >
                  Employment
                </button>
                <button
                  type="button"
                  onClick={() => setActiveSection('statutory')}
                  className={`py-2 rounded-lg transition-all cursor-pointer ${
                    activeSection === 'statutory' ? 'bg-amber-400 text-neutral-950 shadow' : 'text-neutral-400 hover:text-white'
                  }`}
                >
                  Payroll &amp; Tax
                </button>
                <button
                  type="button"
                  onClick={() => setActiveSection('security')}
                  className={`py-2 rounded-lg transition-all cursor-pointer ${
                    activeSection === 'security' ? 'bg-amber-400 text-neutral-950 shadow' : 'text-neutral-400 hover:text-white'
                  }`}
                >
                  Biometrics
                </button>
              </div>

              {/* Tab Content 1: Employment Details */}
              {activeSection === 'employment' && (
                <div className="space-y-2.5 text-xs">
                  <div className="p-3 rounded-xl bg-[#181822] border border-neutral-800/80 flex items-center justify-between">
                    <span className="text-neutral-400">Work Email</span>
                    <span className="text-white font-mono font-semibold">{user.email}</span>
                  </div>

                  <div className="p-3 rounded-xl bg-[#181822] border border-neutral-800/80 flex items-center justify-between">
                    <span className="text-neutral-400">Mobile Phone</span>
                    <span className="text-white font-mono font-semibold">+91 98201 45892</span>
                  </div>

                  <div className="p-3 rounded-xl bg-[#181822] border border-neutral-800/80 flex items-center justify-between">
                    <span className="text-neutral-400">Office Location</span>
                    <span className="text-white font-semibold">SORA Tech Park, BKC Mumbai</span>
                  </div>

                  <div className="p-3 rounded-xl bg-[#181822] border border-neutral-800/80 flex items-center justify-between">
                    <span className="text-neutral-400">Assigned Shift</span>
                    <span className="text-amber-400 font-mono font-bold">{user.shift}</span>
                  </div>

                  <div className="p-3 rounded-xl bg-[#181822] border border-neutral-800/80 flex items-center justify-between">
                    <span className="text-neutral-400">Reporting Manager</span>
                    <span className="text-white font-semibold">Aarav Mehta (Design Director)</span>
                  </div>

                  <div className="p-3 rounded-xl bg-[#181822] border border-neutral-800/80 flex items-center justify-between">
                    <span className="text-neutral-400">Date of Joining</span>
                    <span className="text-white font-mono">{user.joiningDate}</span>
                  </div>
                </div>
              )}

              {/* Tab Content 2: Statutory Compliance & Bank Details */}
              {activeSection === 'statutory' && (
                <div className="space-y-2.5 text-xs">
                  <div className="p-3 rounded-xl bg-[#181822] border border-neutral-800/80 flex items-center justify-between">
                    <span className="text-neutral-400">Primary Bank</span>
                    <span className="text-white font-bold">{user.bank}</span>
                  </div>

                  <div className="p-3 rounded-xl bg-[#181822] border border-neutral-800/80 flex items-center justify-between">
                    <span className="text-neutral-400">Salary Account No.</span>
                    <span className="text-white font-mono font-bold">{user.accountNo}</span>
                  </div>

                  <div className="p-3 rounded-xl bg-[#181822] border border-neutral-800/80 flex items-center justify-between">
                    <span className="text-neutral-400">PAN Card Number</span>
                    <span className="text-emerald-400 font-mono font-bold">{user.pan}</span>
                  </div>

                  <div className="p-3 rounded-xl bg-[#181822] border border-neutral-800/80 flex items-center justify-between">
                    <span className="text-neutral-400">Universal Account No (UAN)</span>
                    <span className="text-white font-mono font-bold">{user.uan}</span>
                  </div>

                  <div className="p-3 rounded-xl bg-[#181822] border border-neutral-800/80 flex items-center justify-between">
                    <span className="text-neutral-400">Provident Fund (PF) No.</span>
                    <span className="text-white font-mono font-semibold">{user.pfNo}</span>
                  </div>

                  <div className="p-3 rounded-xl bg-[#181822] border border-neutral-800/80 flex items-center justify-between">
                    <span className="text-neutral-400">Income Tax Regime</span>
                    <span className="text-amber-400 font-bold">New Tax Regime (FY 2026-27)</span>
                  </div>
                </div>
              )}

              {/* Tab Content 3: Biometrics & Security */}
              {activeSection === 'security' && (
                <div className="space-y-2.5 text-xs">
                  <div className="p-3.5 rounded-2xl bg-[#181822] border border-neutral-800/80 space-y-2">
                    <div className="flex items-center justify-between">
                      <div className="flex items-center gap-2">
                        <Icon name="shield-check" className="w-4 h-4 text-emerald-400" />
                        <span className="font-bold text-white">AI Face Recognition</span>
                      </div>
                      <span className="px-2 py-0.5 rounded-full bg-emerald-500/15 text-emerald-400 text-[10px] font-bold border border-emerald-500/30">
                        Enrolled
                      </span>
                    </div>
                    <p className="text-[11px] text-neutral-400">
                      Biometric facial vectors enrolled on 15 Mar 2023. Used for attendance verification.
                    </p>
                  </div>

                  <div className="p-3.5 rounded-2xl bg-[#181822] border border-neutral-800/80 space-y-2">
                    <div className="flex items-center justify-between">
                      <div className="flex items-center gap-2">
                        <Icon name="map-pin" className="w-4 h-4 text-sky-400" />
                        <span className="font-bold text-white">Authorized Geofence</span>
                      </div>
                      <span className="px-2 py-0.5 rounded-full bg-sky-500/15 text-sky-400 text-[10px] font-bold border border-sky-500/30">
                        BKC Campus (100m)
                      </span>
                    </div>
                    <p className="text-[11px] text-neutral-400">
                      Punches permitted within 100 meters radius of SORA Tech Park, G-Block BKC.
                    </p>
                  </div>

                  <div className="p-3.5 rounded-2xl bg-[#181822] border border-neutral-800/80 space-y-2">
                    <div className="flex items-center justify-between">
                      <div className="flex items-center gap-2">
                        <Icon name="smartphone" className="w-4 h-4 text-amber-400" />
                        <span className="font-bold text-white">Device Binding</span>
                      </div>
                      <span className="px-2 py-0.5 rounded-full bg-amber-500/15 text-amber-400 text-[10px] font-bold border border-amber-500/30">
                        Secured
                      </span>
                    </div>
                    <p className="text-[11px] text-neutral-400">
                      ESS session verified with device hardware signature and SSL encryption.
                    </p>
                  </div>
                </div>
              )}

              {/* Close Button */}
              <button
                type="button"
                onClick={onClose}
                className="w-full py-2.5 px-4 rounded-xl bg-neutral-800 hover:bg-neutral-700 text-white font-bold text-xs transition-all cursor-pointer mt-2"
              >
                Close Profile
              </button>

            </div>
          </div>
        </div>
      );
    }
